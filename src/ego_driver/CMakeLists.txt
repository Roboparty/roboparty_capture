cmake_minimum_required(VERSION 3.16)
project(ego_driver)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(Threads REQUIRED)
set(EGO_SDK_ROOT "" CACHE PATH "Architecture-specific Orbbec SDK root (include/ and lib/)")
find_package(OrbbecSDK CONFIG REQUIRED HINTS "${EGO_SDK_ROOT}/lib")
add_executable(ego_node src/node.cpp src/orbbec_source.cpp)
target_include_directories(ego_node PRIVATE include)
target_link_libraries(ego_node Threads::Threads ob::OrbbecSDK)
ament_target_dependencies(ego_node rclcpp rcl_interfaces std_msgs sensor_msgs)
# Keep the SDK distribution intact: its extensions remain next to its library.
set_property(TARGET ego_node PROPERTY INSTALL_RPATH_USE_LINK_PATH TRUE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ego_node PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS ego_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/read_samples.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
